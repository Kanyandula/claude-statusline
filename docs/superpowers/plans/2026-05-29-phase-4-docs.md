## Per-subcommand `--help` design

A small change to `src/cli/parse-args.js`:

```js
// New optional `usage` param. If present and argv contains --help/-h, the
// usage string is written to stdout and { handled: true, code: 0 } is returned.
// Subcommands short-circuit on `parsed.handled`.
export function parseSubcommandArgs(argv, cmdName, options = {}, usage = null) {
  if (usage && (argv.includes('--help') || argv.includes('-h'))) {
    process.stdout.write(usage);
    return { handled: true, code: 0 };
  }
  // ... existing logic ...
}
```

Each subcommand defines a `USAGE` constant and passes it through:

```js
const USAGE = `\
claude-statusline init [--scope=user|project] [--force]

Wires claude-statusline into Claude Code by adding a statusLine entry to
~/.claude/settings.json and creating an empty config file. Defaults apply.

Options:
  --scope=user|project   Where to write the config file (default: user)
  --force                Overwrite an existing statusLine entry
  --help, -h             Print this help
`;

export function run(argv) {
  const parsed = parseSubcommandArgs(argv, 'init', { force: { type: 'boolean', default: false } }, USAGE);
  if (parsed.handled) return parsed.code;
  if (parsed.error) { /* ... */ }
  // ...
}
```

## index.js help text scoping

Current help lists `--force` and `--keep-config` under "Global options" — they're per-command. Split into "Global options" (just `--scope`) and per-command lines:

```
Commands:
  init        Wire claude-statusline into Claude Code              [--force]
  layout      Switch layout: single | two-line
  enable      Turn on a field
  disable     Turn off a field
  set         Set a config key by dotted path
  get         Print effective config as JSON
  preview     Render the statusline once with sample data          [--live]
  reset       Restore defaults (deletes config file)
  uninstall   Remove statusLine from settings.json                 [--keep-config]
  render      Render statusline from stdin (invoked by Claude Code)

Global options:
  --scope=user|project   Which file to read/write (default: user)
  --help, -h             Print command-specific help

Run `claude-statusline <command> --help` for command-specific options.
```

The `COMMANDS` registry gains a `flags` field with the per-command flag list; `buildHelp` formats them into the same line.

- [ ] **Step 2: Refactor `uninstall.js`**

Replace raw `parseArgs` with `parseSubcommandArgs` using `includeScope: false`:

```js
import { resolveSettingsPath, resolveUserConfigPath } from './paths.js';
import { removeStatusLine } from './settings.js';
import { deleteConfigFile } from './config-file.js';
import { existsSync } from 'node:fs';
import { parseSubcommandArgs } from './parse-args.js';

const USAGE = `\
claude-statusline uninstall [--keep-config]

Removes the statusLine entry from ~/.claude/settings.json and (unless
--keep-config) deletes ~/.claude/claude-statusline.json. Project-level
configs are not touched.

Options:
  --keep-config   Preserve ~/.claude/claude-statusline.json
  --help, -h      Print this help
`;

export function run(argv) {
  const parsed = parseSubcommandArgs(argv, 'uninstall', {
    'keep-config': { type: 'boolean', default: false },
  }, USAGE, { includeScope: false });
  if (parsed.handled) return parsed.code;
  if (parsed.error) { process.stderr.write(`${parsed.error}\n`); return parsed.code; }

  const keepConfig = parsed.values['keep-config'];
  const settingsPath = resolveSettingsPath();
  const removed = removeStatusLine(settingsPath);

  const configPath = resolveUserConfigPath();
  const configExisted = existsSync(configPath);
  if (!keepConfig) deleteConfigFile(configPath);

  const lines = [];
  lines.push(removed
    ? `Removed statusLine from ${settingsPath}.`
    : `No statusLine entry in ${settingsPath} (already uninstalled).`);
  if (keepConfig && configExisted) lines.push(`Kept ${configPath} (--keep-config).`);
  else if (!keepConfig && configExisted) lines.push(`Deleted ${configPath}.`);
  else if (!keepConfig && !configExisted) lines.push(`No config file at ${configPath} (already gone).`);
  lines.push('Restart Claude Code for the change to take effect.');
  process.stdout.write(lines.join('\n') + '\n');
  return 0;
}
```

Note: this requires `parseSubcommandArgs` to accept a 5th param `{ includeScope }`. Update the helper signature in Task 1's commit, OR fold this into Task 3 if cleaner.
